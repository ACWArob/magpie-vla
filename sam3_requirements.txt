# ============================================================
# sam3_env — Python 3.12 isolated venv for SAM3
# ============================================================
# SAM3 requires Python 3.12+ and cannot coexist with ROS Python 3.10.
# This file is for ~/sam3_env ONLY — do NOT pip install this into the system.
#
# One-time setup:
#   python3.12 -m venv ~/sam3_env
#   ~/sam3_env/bin/pip install torch torchvision --index-url https://download.pytorch.org/whl/cu128
#   cd /home/user/sam3 && ~/sam3_env/bin/pip install -e .
#   ~/sam3_env/bin/pip install -r sam3_requirements.txt
#
# HuggingFace access (gated model — one-time):
#   HF_HUB_OFFLINE=0 ~/sam3_env/bin/huggingface-cli login
#   (paste token from huggingface.co/settings/tokens)
#   Visit huggingface.co/facebook/sam3 and request access if not already granted.
#
# RTX 2070 (Turing / SM 7.5) note:
#   SAM3 source has been globally patched: torch.bfloat16 → torch.float32
#   (Turing does not support bfloat16; float32 is used throughout instead)
#   Patch location: /home/user/sam3/sam3/**/*.py
